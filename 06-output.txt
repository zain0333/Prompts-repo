import math

# Ask the user to enter a number
number = int(input("Enter a number: "))

# Check if the number is valid
if number < 0:
    print("Invalid input! Please enter 0 or a positive number.")
else:
    # Calculate factorial
    result = math.factorial(number)

    # Display the result
    print(f"\nThe factorial of {number} is {result}")

    # Ask for feedback
    print("\nHow was this program?")
    print("1. Awesome")
    print("2. Good")
    print("3. Neutral")
    print("4. Bad")

    feedback = input("Enter your feedback: ").strip().lower()

    if feedback == "1" or feedback == "awesome":
        print("😊 Thank you! I'm glad you loved it.")
    elif feedback == "2" or feedback == "good":
        print("🙂 Thank you! I'm happy you liked it.")
    elif feedback == "3" or feedback == "neutral":
        print("😐 Thanks for your feedback. I'll try to improve.")
    elif feedback == "4" or feedback == "bad":
        print("☹️ Thank you for your honesty. I'll work on making it better.")
    else:
        print("🙏 Thank you for your feedback!")
